```python
import sqlite3

conn = sqlite3.connect('demo.db')        # 连接数据库文件，不存在就新建
cur = conn.cursor()

# 建表：表已存在就跳过
cur.execute('''
    CREATE TABLE IF NOT EXISTS bookmarks (
        id    INTEGER PRIMARY KEY AUTOINCREMENT,
        title VARCHAR(100) NOT NULL,
        url   VARCHAR(500) NOT NULL UNIQUE
    )
''')

# 增：数据用占位符 ? 传入，不拼进 SQL 字符串
presets = [('Flask 官方文档', 'https://flask.palletsprojects.com/'),
           ('阿里云开发者社区', 'https://developer.aliyun.com/')]
for title, url in presets:
    try:
        cur.execute('INSERT INTO bookmarks (title, url) VALUES (?, ?)', (title, url))
    except sqlite3.IntegrityError as e:    # 违反约束：数据库拒绝写入
        print('已存在，跳过：', e)
conn.commit()                              # 写操作提交后才保存到文件

# 查：逐行取出查询结果，每行是一个元组
for row in cur.execute('SELECT id, title, url FROM bookmarks'):
    print(row)
```

12. **写配置与模型**。在 `app.py` 里，把 `from flask import Flask` 那一行起，到 `next_id = 4` 那一行止，整段换成下面的代码，保存：

```python
from flask import Flask, render_template, request, redirect, url_for, flash
from flask_sqlalchemy import SQLAlchemy
from validators import validate_bookmark

app = Flask(__name__)
# flash 存在会话里，会话要用密钥签名；密钥从环境变量读，代码里只留开发用的默认值
app.secret_key = os.environ.get('SECRET_KEY', 'dev-secret-key')
# 数据库地址从环境变量读；没设时用 SQLite 文件（实际放在 instance/bookmarks.db）
app.config['SQLALCHEMY_DATABASE_URI'] = os.environ.get('DATABASE_URL', 'sqlite:///bookmarks.db')
db = SQLAlchemy(app)


class Bookmark(db.Model):
    """书签模型：类对应表，对象对应一行，属性对应一列"""
    __tablename__ = 'bookmarks'                          # 表名；不写时默认是类名小写 bookmark
    id = db.Column(db.Integer, primary_key=True)         # 主键，数据库自动编号，代替原来的 next_id
    title = db.Column(db.String(100), nullable=False)    # 长度与 validators.py 的上限一致
    url = db.Column(db.String(500), nullable=False)


PRESETS = [
    ('Flask 官方文档', 'https://flask.palletsprojects.com/'),
    ('Jinja2 模板文档', 'https://jinja.palletsprojects.com/'),
    ('阿里云开发者社区', 'https://developer.aliyun.com/'),
]

# 在请求之外操作数据库，要先进入应用上下文；视图函数里不用写
with app.app_context():
    db.create_all()                                      # 按模型建表；表已存在就跳过，不会改表
    if Bookmark.query.count() == 0:                      # 表为空（第一次运行）时插入预设三条
        db.session.add_all([Bookmark(title=t, url=u) for t, u in PRESETS])
        db.session.commit()
```

13. **换掉五条路由**。在 `app.py` 里，把 `# 接口表「查看列表」` 那一行起，到 `delete_bookmark` 的最后一行止，整段换成下面的代码，保存。`# 接口表「健康检查」` 以下不动。

```python
# 接口表「查看列表」：视图函数只准备数据，页面长什么样由模板决定
@app.route('/')
def index():
    bookmarks = Bookmark.query.all()                     # SELECT * FROM bookmarks
    return render_template('index.html', bookmarks=bookmarks)


# 接口表「打开添加表单」「提交新书签」：同一个地址，GET 显示，POST 处理
@app.route('/add', methods=['GET', 'POST'])
def add_bookmark():
    if request.method == 'POST':
        # 按表单的 name 取值；.get 给默认值，字段缺失也不出错；strip 去掉首尾空格
        title = request.form.get('title', '').strip()
        url = request.form.get('url', '').strip()

        errors = validate_bookmark(title, url)
        if errors:                                       # 失败：报错，留在表单上，回填已填内容
            for e in errors:
                flash(e, 'error')
            return render_template('add.html', title=title, url=url)

        db.session.add(Bookmark(title=title, url=url))   # 登记到会话
        db.session.commit()                              # 提交：生成 INSERT 写入数据库
        flash('书签添加成功！', 'success')
        return redirect(url_for('index'))   # PRG：浏览器接着 GET /，刷新只重发这个 GET
    return render_template('add.html')


def find_bookmark(bookmark_id):
    """按 id 找书签，找不到返回 None；编辑、删除都用它"""
    return db.session.get(Bookmark, bookmark_id)        # 按主键查询


# 接口表「打开编辑表单」「提交修改」：先找到书签，其余与添加相同
@app.route('/edit/<int:bookmark_id>', methods=['GET', 'POST'])
def edit_bookmark(bookmark_id):
    bookmark = find_bookmark(bookmark_id)
    if bookmark is None:                               # 找不到：提示后回首页
        flash('书签不存在', 'error')
        return redirect(url_for('index'))
    if request.method == 'POST':
        title = request.form.get('title', '').strip()
        url = request.form.get('url', '').strip()

        errors = validate_bookmark(title, url)
        if errors:                                     # 失败：回填用户刚填的，不是原来的
            for e in errors:
                flash(e, 'error')
            return render_template('edit.html', title=title, url=url)

        bookmark.title = title                         # 改对象的属性
        bookmark.url = url
        db.session.commit()                            # 提交：生成 UPDATE
        flash('书签更新成功！', 'success')
        return redirect(url_for('index'))
    # GET：表单先填好这条书签的当前值
    return render_template('edit.html', title=bookmark.title, url=bookmark.url)


# 接口表「删除」：只收 POST，按钮放在 index.html 的表单里
@app.route('/delete/<int:bookmark_id>', methods=['POST'])
def delete_bookmark(bookmark_id):
    bookmark = find_bookmark(bookmark_id)
    if bookmark is None:
        flash('书签不存在', 'error')
    else:
        db.session.delete(bookmark)                    # 登记删除
        db.session.commit()                            # 提交：生成 DELETE
        flash('书签已删除', 'success')
    return redirect(url_for('index'))
```

- **提示**：列类型用 `db.DateTime`，`default` 传函数名而不是调用结果；排序用 `order_by` 与 `desc()`。改了表结构，按上面的规则删库重建。

**E2　网址不能重复**

- **任务**：`url` 列加唯一约束；添加重复网址时停在表单页，提示「该网址已收藏」，不出现错误页。
- **提示**：列参数里有 `unique`；违反约束时 `commit` 抛出 `sqlalchemy.exc.IntegrityError`，捕获它，先 `rollback()` 再回填表单。改了表结构，按上面的规则删库重建。
- **验证**：添加预设里已有的网址，表单页显示提示，没有 500 错误页；之后添加新网址仍能成功。

**E3　按标题搜索**

- **任务**：首页支持地址 `/?q=关键词`，只列标题含关键词的书签；不带 `q` 时列出全部。
- **提示**：用 `request.args.get` 取查询参数；`filter` 配合列的 `like`，关键词前后加 `%`；没有关键词时不加条件。
- **验证**：`/?q=Flask` 只列标题含 Flask 的书签；`/` 仍列全部；搜不存在的词时显示「共 0 条书签」。

## 出错了先查这里

| 步骤 | 现象 | 处理 |
|---|---|---|
| 3 | `git status` 列出了 `demo.db` | 第 3 步没做或写错，补做后再提交 |
| 5 | `can't open file 'sql_demo.py'` | 终端不在 `~/work/cloud-bookmark`，先执行 `cd ~/work/cloud-bookmark` |
| 11、30 | `No module named 'flask_sqlalchemy'` | 没装或虚拟环境没激活：`source ~/work/venv/bin/activate`，再执行 `pip install -r requirements.txt` |
| 14 | 报 `NameError`，提到 `next_id` 或 `bookmarks` | 旧写法还留着：对照第 12、13 步，把 `next_id`、`bookmarks.append`、`bookmarks.remove` 换掉 |
| 19 | 重启后新添加的书签没有了 | 添加的路由里漏了 `db.session.commit()`，对照第 13 步补上 |
| 21 | `AttributeError`，提到 `bookmarks` | 第 21 步没替换，测试里还是 `bookmarks.clear()` |
| 22 | 测试之后开发用的书签被清空 | `tests/conftest.py` 没有生效，测试连到了开发库：文件要放在 `tests/` 里，名字叫 `conftest.py` |
| 27 | 报 `port is already allocated` | 3306 端口被占用：`docker ps` 找到占用的容器，`docker stop` 它；或停掉本机的 MySQL 服务。不要改端口号 |
| 27 | 报 `container name "/cb-mysql" is already in use` | 容器已存在：`docker start cb-mysql` 直接启动它 |
| 27 | `docker run` 没带 `-v`，删过容器后书签没了 | 没有 `-v`，数据放在容器里，容器一删就丢。带上 `-v cb-mysql-data:/var/lib/mysql`，数据存进卷，删了容器重建数据还在 |
| 27 | 提示找不到镜像 `mysql:8.0` 或一直拉不下来 | 镜像应在第8讲课后拉取；拉不下来找老师拷贝镜像文件 |
| 32 | `ModuleNotFoundError: No module named 'pymysql'` | 第 29、30 步没做完，或终端 1 的虚拟环境没激活 |
| 32 | `RuntimeError`，提到 `cryptography` | 缺 `cryptography`：第 29 步的两行都要加，再执行第 30 步 |
| 32 | `OperationalError`，提到 `Lost connection to MySQL server` 或 `Can't connect to MySQL server` | 先 `docker ps` 看 `cb-mysql` 是不是 `Up`；是的话 MySQL 还没就绪，按第 28 步等到 `port: 3306` 那行出现再启动服务器 |
| 32 | `Access denied for user 'cb'` | 地址里的账号密码写错：应是 `cb`、`cb-pass` |
| 32 | 首页还是原来的数据，不是新建的三条 | 终端 1 没设 `DATABASE_URL`：重做第 31 步，用 `echo $DATABASE_URL` 检查 |
